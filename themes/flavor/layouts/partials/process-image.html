{{- $image := .image | default . -}} {{- $op := "" -}} {{- $spec := "" -}} {{- $q := 80 -}} {{- if reflect.IsMap . -}} {{- with index . "op" }}{{ $op = . }}{{ end -}} {{- with index . "spec" }}{{ $spec = . }}{{ end -}} {{- with index . "q" }}{{ $q = int . }}{{ end -}} {{- end -}} {{- $rendered := $image -}} {{- $sub := $image.MediaType.SubType -}} {{- if or (eq $sub "gif") (eq $sub "svg+xml") -}} {{- else if and $op $spec -}} {{- $specStr := printf "%s webp q%d" $spec $q -}} {{- $tr := false -}} {{- if eq $op "fill" -}} {{- $tr = try ($image.Fill $specStr) -}} {{- else if eq $op "resize" -}} {{- $tr = try ($image.Resize $specStr) -}} {{- else if eq $op "fit" -}} {{- $tr = try ($image.Fit $specStr) -}} {{- end -}} {{- if and $tr (not $tr.Err) -}} {{- $rendered = $tr.Value -}} {{- end -}} {{- else if or (gt $image.Width 1600) (gt $image.Height 1600) -}} {{- $tr := try ($image.Fit (printf "1600x1600 webp q%d" $q)) -}} {{- if not $tr.Err }}{{ $rendered = $tr.Value }}{{ end -}} {{- else if ne $sub "webp" -}} {{- $tr := try ($image.Process (printf "webp q%d" $q)) -}} {{- if not $tr.Err }}{{ $rendered = $tr.Value }}{{ end -}} {{- end -}} {{- return $rendered -}}
